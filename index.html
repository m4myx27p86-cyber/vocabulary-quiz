let quizData = []; let currentQuestion = 0; let score = 0; fetch("stock_3000_section1.csv") .then(response => response.text()) .then(data => { const rows = data.trim().split("\n").slice(1); quizData = rows.map(row => { const cols = row.split(","); return { id: cols[0], // A列 word: cols[1], // B列 correct: cols[2], // C列 choices: [ cols[2], // Correct cols[3], // Wrong1 cols[4], // Wrong2 cols[5] // Wrong3 ], points: cols[6] // G列（今は表示だけでもOK） }; }); shuffleArray(quizData); showQuestion(); }); function showQuestion() { if (currentQuestion >= quizData.length) { document.getElementById("quiz-container").innerHTML = `

Quiz Finished!

Your Score: ${score} / ${quizData.length}

`; return; } const q = quizData[currentQuestion]; document.getElementById("question").innerText = `No. ${q.id} / Question ${currentQuestion + 1}: What is the meaning of "${q.word}"?`; document.getElementById("result").innerText = ""; const choicesDiv = document.getElementById("choices"); choicesDiv.innerHTML = ""; let shuffledChoices = [...q.choices]; shuffleArray(shuffledChoices); shuffledChoices.forEach(choice => { const btn = document.createElement("button"); btn.innerText = choice; btn.onclick = function () { if (choice === q.correct) { document.getElementById("result").innerText = "✅ Correct!"; score += Number(q.points); } else { document.getElementById("result").innerText = `❌ Wrong! Correct answer: ${q.correct}`; } disableChoices(); }; choicesDiv.appendChild(btn); }); document.getElementById("score").innerText = `Score: ${score}`; }
